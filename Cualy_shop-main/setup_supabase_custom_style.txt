-- 1. Agregar columnas de estilo a la tabla de productos
ALTER TABLE productos 
ADD COLUMN IF NOT EXISTS custom_html TEXT,
ADD COLUMN IF NOT EXISTS custom_css TEXT;

-- 2. Crear la tabla para guardar los temas reutilizables
CREATE TABLE IF NOT EXISTS temas (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre TEXT NOT NULL,
  html TEXT,
  css TEXT,
  color TEXT DEFAULT '#00AEEF',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Habilitar permisos (RLS)
-- Nota: Esto permite lectura y escritura a cualquiera. 
-- Si quieres más seguridad, puedes ajustar las políticas.
ALTER TABLE temas ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Acceso publico temas" ON temas;
CREATE POLICY "Acceso publico temas" ON temas FOR ALL USING (true);
